# Tests de contrato

> [!NOTE]
> Esta es una funcionalidad experimental que aún está en desarrollo activo.
> No hay tests e2e implementados todavía. Esta página describe la estructura
> prevista.

Los tests de contrato verifican que los endpoints cumplen la especificación
JSON:API tanto en el formato de las respuestas exitosas como en el de los
errores. A diferencia de los tests de integración, estos tests se ejecutan
contra la aplicación completa montada con `httpx.AsyncClient`.

## Estructura de tests

```
tests/
├── e2e/
│   ├── __init__.py
│   ├── conftest.py
│   ├── test_auth_endpoints.py
```

## Validación de estructura JSON:API

Cada test verifica que la respuesta tenga la estructura correcta:

```python
class TestAuthContract:
    """Valida que los endpoints auth cumplan el contrato JSON:API."""

    @pytest.mark.asyncio
    async def test_register_returns_201_with_token_document(
        self, client
    ):
        payload = {
            "data": {
                "type": "users",
                "attributes": {
                    "email": "user@example.com",
                    "password": "StrongP@ss1",
                },
            }
        }

        response = await client.post(
            "/v1/auth/register",
            json=payload,
            headers={
                "Content-Type": "application/vnd.api+json",
                "Accept": "application/vnd.api+json",
            },
        )

        assert response.status_code == 201
        assert response.headers["Content-Type"] == \
            "application/vnd.api+json"

        body = response.json()
        assert "data" in body
        assert body["data"]["type"] == "tokens"
        assert "attributes" in body["data"]
        assert "access_token" in body["data"]["attributes"]
        assert "refresh_token" in body["data"]["attributes"]

    @pytest.mark.asyncio
    async def test_register_with_weak_password_returns_422(
        self, client
    ):
        payload = {
            "data": {
                "type": "users",
                "attributes": {
                    "email": "user@example.com",
                    "password": "weak",
                },
            }
        }

        response = await client.post(
            "/v1/auth/register",
            json=payload,
            headers={
                "Content-Type": "application/vnd.api+json",
                "Accept": "application/vnd.api+json",
            },
        )

        assert response.status_code == 422
        body = response.json()
        assert "errors" in body
        assert body["errors"][0]["code"] == "ERR_WEAK_PASSWORD"
        assert "source" in body["errors"][0]
        assert body["errors"][0]["source"]["pointer"] == \
            "/data/attributes/password"
```

## Flujo completo de autenticación

```python
class TestAuthFlow:
    @pytest.mark.asyncio
    async def test_full_auth_flow(self, client):
        # 1. Registro
        register_payload = {
            "data": {
                "type": "users",
                "attributes": {
                    "email": "flow@example.com",
                    "password": "StrongP@ss1",
                },
            }
        }
        reg_resp = await client.post(
            "/v1/auth/register",
            json=register_payload,
            headers={
                "Content-Type": "application/vnd.api+json",
                "Accept": "application/vnd.api+json",
            },
        )
        assert reg_resp.status_code == 201
        tokens = reg_resp.json()["data"]["attributes"]
        access_token = tokens["access_token"]
        refresh_token = tokens["refresh_token"]

        # 2. Obtener perfil
        me_resp = await client.get(
            "/v1/auth/me",
            headers={
                "Authorization": f"Bearer {access_token}",
                "Accept": "application/vnd.api+json",
            },
        )
        assert me_resp.status_code == 200
        me_body = me_resp.json()
        assert me_body["data"]["type"] == "users"
        assert me_body["data"]["attributes"]["email"] == \
            "flow@example.com"

        # 3. Refrescar tokens
        refresh_payload = {
            "data": {
                "type": "tokens",
                "attributes": {"refresh_token": refresh_token},
            }
        }
        ref_resp = await client.post(
            "/v1/auth/refresh",
            json=refresh_payload,
            headers={
                "Content-Type": "application/vnd.api+json",
                "Accept": "application/vnd.api+json",
            },
        )
        assert ref_resp.status_code == 200

        # 4. Cerrar sesión
        new_access = ref_resp.json()["data"]["attributes"][
            "access_token"
        ]
        logout_resp = await client.post(
            "/v1/auth/logout",
            headers={
                "Authorization": f"Bearer {new_access}",
                "Content-Type": "application/vnd.api+json",
                "Accept": "application/vnd.api+json",
            },
        )
        assert logout_resp.status_code == 204
```

## Verificación de cabeceras de contenido

```python
class TestContentTypeContract:
    @pytest.mark.asyncio
    async def test_wrong_content_type_returns_415(self, client):
        payload = {
            "data": {
                "type": "users",
                "attributes": {
                    "email": "user@example.com",
                    "password": "StrongP@ss1",
                },
            }
        }

        response = await client.post(
            "/v1/auth/register",
            json=payload,
            headers={
                "Content-Type": "application/json",
                "Accept": "application/vnd.api+json",
            },
        )

        assert response.status_code == 415
        assert response.headers["Content-Type"] == \
            "application/vnd.api+json"
        body = response.json()
        assert body["errors"][0]["code"] == \
            "ERR_UNSUPPORTED_MEDIA_TYPE"

    @pytest.mark.asyncio
    async def test_wrong_accept_header_returns_406(self, client):
        response = await client.get(
            "/v1/auth/me",
            headers={"Accept": "text/plain"},
        )

        assert response.status_code == 406
        body = response.json()
        assert body["errors"][0]["code"] == "ERR_NOT_ACCEPTABLE"
```
